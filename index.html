01

Connect to Google Sheet

live data via Apps Script

One-time setup in your new "Spun Master Sheet new":
1. Open "Spun Master Sheet new" → Extensions → Apps Script.
2. Delete any code there and paste the script below.
3. Deploy → New deployment → type Web app → Execute as Me, Access Anyone → Deploy.
4. Copy the Web App URL it gives you and paste it into the box on the left, then click Connect.
The old Web App URL pointed at the previous master sheet, so a fresh deployment (and a new URL) is needed here.
The script now reads every month tab (July 2026, June 2026, …) in the spreadsheet automatically and combines them — no need to point it at one sheet name. 

function doGet(e) {
  // "Spun Master Sheet new" stores data in separate month tabs
  // (e.g. "July 2026", "June 2026", ...). This reads every tab
  // that has the expected header row and combines them into one
  // JSON array, so the dashboard sees all months at once.
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheets = ss.getSheets();
  var rows = [];

  sheets.forEach(function (sheet) {
    var data = sheet.getDataRange().getValues();
    if (data.length < 2) return; // no data rows

    var headers = data[0];
    // only treat a tab as a data tab if it has the expected columns
    if (headers.indexOf('Date') === -1 || headers.indexOf('SN') === -1) return;

    for (var i = 1; i < data.length; i++) {
      var sn = data[i][headers.indexOf('SN')];
      if (sn === '' || sn === null || sn === undefined) continue; // skip blank rows

      var row = {};
      for (var j = 0; j < headers.length; j++) {
        var val = data[i][j];
        if (val instanceof Date) {
          val = Utilities.formatDate(val, Session.getScriptTimeZone(), 'dd-MMM-yyyy');
        }
        row[headers[j]] = val;
      }
      rows.push(row);
    }
  });

  return ContentService.createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}
